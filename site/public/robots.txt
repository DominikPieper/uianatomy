# https://www.rfc-editor.org/rfc/rfc9309
# All paths are public documentation. No restricted areas.
#
# Content-Signal: declares AI usage preferences per
# https://contentsignals.org/ and
# https://datatracker.ietf.org/doc/draft-romm-aipref-contentsignals/
#   ai-train=no   — content may not be used for training corpora
#   search=yes    — indexable by search engines
#   ai-input=yes  — agents may consume content per request
#
# The default (User-agent: *) keeps general search crawlers focused on
# canonical HTML pages so duplicates (.md sidecars, /api/*.json, OG
# images) do not dilute ranking signal. Per-AI-bot blocks below override
# that to grant Perplexity, ChatGPT, Claude, Gemini, and Common Crawl
# explicit access to the JSON catalogue and Markdown corpus, since
# those endpoints are designed for AI ingestion. The disallowed paths
# remain reachable for AI agents at request time
# (Content-Signal: ai-input=yes); robots.txt only gates crawler
# indexing, not on-demand fetching.

User-agent: *
Content-Signal: ai-train=no, search=yes, ai-input=yes
Allow: /
Disallow: /api/
Disallow: /og/
Disallow: /pagefind/
Disallow: /llms.txt
Disallow: /llms-full.txt
Disallow: /*.md$
